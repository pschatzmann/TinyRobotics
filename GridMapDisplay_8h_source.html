7/**
8 * @class GridMapDisplay
9 * @ingroup communication
10 * @brief A class that visualizes a GridMap using the TinyGPU display API.
11 * This class provides a way to visualize the state of a GridMap on a TinyGPU
12 * display. It sets up a callback to update the display whenever a cell state
13 * changes, and allows for customization of the color mapping and vehicle
14 * sprite.
15 * @note Dependency: https://pschatzmann.github.io/TinyGPU
16 */
17
18template <typename TRGB = RGB565, typename StateT = CellState,
19 typename T = DistanceM>
20class GridMapDisplay {
21 public:
22 GridMapDisplay(GridMap<StateT, T> map, uint32_t widthPixels,
23 uint32_t heightPixels)
24 : p_gridMap(&map), widthPixels(widthPixels), heightPixels(heightPixels) {
25 deltaXPixels = widthPixels / map.getXCount();
26 deltaYPixels = heightPixels / map.getYCount();
27 }
28
29 /// Initialize the GPU and set up callbacks
30 bool begin() {
31 framebuffer.resize(widthPixels, heightPixels);
32 if (!framebuffer.begin()) {
33 // Serial.println("Failed to initialize TinyGPU framebuffer");
34 return false;
35 }
36
37 // Set the callback to update the GPU whenever a cell state changes
38 p_gridMap->setCellUpdateCallback([this](GridMap<StateT, T>& map, int cx,
39 int cy, const Coordinate<T>& coord,
40 CellState state) {
41 this->updateGridMap(map, cx, cy, coord, state);
42 });
43 updateGPU();
44 return true;
45 }
46
47 /// Clean up resources and remove callbacks
48 void end() {
49 framebuffer.end();
50 // Remove callback to avoid dangling pointer
51 p_gridMap->setCellUpdateCallback(nullptr);
52 }
53
54 /// Set the position and orientation of the vehicle on the grid map
55 void displayVehicleAt(T x, T y, float angleDegrees) {
56 // Account for grid origin
57 T relX = x - p_gridMap->getOriginX();
58 T relY = y - p_gridMap->getOriginY();
59 int posX = static_cast<int>(relX / p_gridMap->getResolution());
60 int posY = static_cast<int>(relY / p_gridMap->getResolution());
61 // Center the sprite in the cell
62 float pixelX = (posX + 0.5f) * deltaXPixels;
63 float pixelY = (posY + 0.5f) * deltaYPixels;
64 if (p_sprite_info == nullptr) {
65 p_sprite_info = &framebuffer.addSprite(pixelX, pixelY, vehicleSprite,
66 getColorForState(CellState::OCCUPIED));
67 } else {
68 p_sprite_info->setPosition(pixelX, pixelY);
69 }
70 p_sprite_info->setRotation(angleDegrees);
71 }
72
73 /// Define a custom color mapping for cell states (optional)
74 void setColorCallback(TRGB (*cb)(StateT state)) { getColorCB = cb; }
75
76 /// Define a custom sprite for the vehicle (optional)
77 void setSprite(const Sprite<TRGB>& newSprite) {
78 vehicleSprite = newSprite;
79 // If the sprite is already on the GPU, update it
80 if (p_sprite_info != nullptr) {
81 float x = p_sprite_info->getX();
82 float y = p_sprite_info->getY();
83 float angle = p_sprite_info->getRotation();
84 framebuffer.removeSprite(p_sprite_info);
85 p_sprite_info = &framebuffer.addSprite(x, y, vehicleSprite,
86 getColorForState(CellState::OCCUPIED));
87 p_sprite_info->setRotation(angle);
88 }
89 }

91 /// Access the underlying TinyGPU framebuffer for direct drawing (optional)
92 FrameBuffer<TRGB>& getFrameBuffer() { return framebuffer; }
93
94 protected:
95 GridMap<StateT, T>* p_gridMap;
96 FrameBuffer<TRGB> framebuffer;
97 int widthPixels = 0; // Display width in pixels
98 int heightPixels = 0; // Display height in pixels
99 uint32_t deltaXPixels = 0; // Pixels per grid cell in X direction
100 uint32_t deltaYPixels = 0; // Pixels per grid cell in Y direction
101 TRGB (*getColorCB)(StateT state) = nullptr;

103 {
104 0b00000000,
105 0b00011000,
106 0b00111100,
107 0b01111110,
108 0b00011000,
109 0b00011000,

111 0b00011000,
112 0b00000000,
113 });
114 FrameBuffer<TRGB>::SpriteInfo* p_sprite_info = nullptr;

116 TRGB getColorForState(StateT state) {
117 // If a custom color callback is set, use it to get the color for the state
118 if (getColorCB != nullptr) {
119 return getColorCB(state);
120 }
121 // Default color mapping for CellState
122 if constexpr (std::is_same<StateT, CellState>::value) {
123 switch (state) {
124 case CellState::FREE:
125 return TRGB(0, 255, 0); // Green
126 case CellState::OCCUPIED:
127 return TRGB(255, 0, 0); // Red
128 case CellState::UNKNOWN:
129 return TRGB(128, 128, 128); // Gray
130 default:
131 return TRGB(0, 0, 0); // Black for invalid state
132 }
133 }
134 }
135
136 /// update the entire GPU grid map based on the current state of the GridMap
137 void updateGPU() {
138 // For each cell in the grid map, update the corresponding pixels in the GPU
139 for (int y = 0; y < p_gridMap->getYCount(); ++y) {
140 for (int x = 0; x < p_gridMap->getXCount(); ++x) {
141 CellState state;
142 if (p_gridMap->getCell(x, y, state)) {
143 updateGridMap(*p_gridMap, x, y, p_gridMap->toWorld(x, y), state);
144 }
145 }
146 }
147 }
148
149 /// Callback function to update the GPU when a cell state changes
150 void updateGridMap(GridMap<StateT, T>& map, int cx, int cy,
151 const Coordinate<T>& coord, CellState state) {
152 TRGB color = getColorForState(state);
153 int px = cx * deltaXPixels;
154 int py = cy * deltaYPixels;
155 // Fill the cell with the appropriate color
156 framebuffer.fillRect(px, py, deltaXPixels, deltaYPixels, color);
157 // Draw border (black)
158 framebuffer.drawRect(px, py, deltaXPixels, deltaYPixels, TRGB(0, 0, 0));
159 }
160};
