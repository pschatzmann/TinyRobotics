3/**
4 * @file StepperMotorLinux.h
5 * @ingroup motors
6 * @brief Stepper driver for Raspberry Pi (Linux) using hardware PWM.
7 *
8 * Only continuous speed mode is supported: the step pin is driven with a
9 * PWM signal whose frequency is the step rate, and the direction pin selects
10 * the rotation direction. Counted moves (moveRevolutions, moveDistance) and
11 * acceleration ramps are NOT supported, because sysfs PWM cannot count steps.
12 *
13 * Hardware PWM is available only on BCM pins 12, 13, 18 and 19, which share
14 * two PWM channels. Do not use the same channel for a servo and a stepper.
15 */
16

24/**
25 * @class StepperMotorLinux
26 * @ingroup motors
27 * @brief Continuous-speed stepper driver for Raspberry Pi using PWM step pulses.
28 *
29 * Usage:
30 * @code
31 * StepperMotorLinux<float> stepper(0);
32 * stepper.setPins(18, 23, 24); // step (PWM), direction, enable
33 * stepper.setMaxSpeed(1000); // steps per second at 100%
34 * stepper.begin();
35 * stepper.setValuePercent(50); // 500 steps/s forward
36 * stepper.end(); // stop pulses
37 * @endcode
38 */
39template <typename T = float>
40class StepperMotorLinux : public IMotor<T> {
41 public:
42 StepperMotorLinux(uint8_t id = 0) : gpio(arduino::RPI.getGPIO()) {
43 this->setID(id);
44 }
45
46 /// Set the step (must be a hardware PWM pin), direction and enable pins.
47 /// The enable pin is active LOW, as on common A4988/DRV8825 drivers.
48 void setPins(int stepPin, int dirPin, int enablePin = -1) {
49 pinStep = stepPin;
50 pinDir = dirPin;
51 pinEnable = enablePin;
52 is_pins_set = true;
53 }
54
55 /// Steps per second at 100 percent speed.
56 void setMaxSpeed(float stepsPerSec) { maxSpeedHz = stepsPerSec; }
57
58 /// Not supported on Linux: the PWM output has no acceleration ramp. The value
59 /// is stored for API compatibility only.
60 void setAccelerationMs(uint16_t ms) { accelerationMs = ms; }
61
62 /// Steps per full revolution (informational, used by unit conversions).
63 void setStepsPerRevolution(int steps) { stepsPerRevolution = steps; }
64
65 /// Configure the pins. Returns false if the pins were not set.
66 bool begin() override {
67 if (!is_pins_set || gpio == nullptr) return false;
68 gpio->pinMode(pinStep, OUTPUT);
69 gpio->pinMode(pinDir, OUTPUT);
70 if (pinEnable != -1) {
71 gpio->pinMode(pinEnable, OUTPUT);
72 gpio->digitalWrite(pinEnable, LOW); // enable driver
73 }
74 return true;
75 }
76
77 /// Continuous speed in percent (-100 to 100). Positive is forward.
78 bool setValuePercent(T percent) override {
79 if (!is_pins_set || gpio == nullptr) return false;
80 lastValuePercent = constrain(percent, -100.0f, 100.0f);
81 if (lastValuePercent == 0) {
82 gpio->noTone(pinStep);
83 return true;
84 }
85 gpio->digitalWrite(pinDir, lastValuePercent > 0 ? HIGH : LOW);
86 float hz = maxSpeedHz * fabs((float)lastValuePercent) / 100.0f;
87 if (hz < 1.0f) hz = 1.0f; // sysfs PWM cannot run below 1 Hz
88 gpio->tone(pinStep, (unsigned int)hz);
89 return true;
90 }
91
92 T getValuePercent() const override { return lastValuePercent; }
93
94 /// Stop the step pulses.
95 void end() override {
96 if (gpio != nullptr && is_pins_set) gpio->noTone(pinStep);
97 lastValuePercent = 0;
98 }
99
100 bool isPinsSet() const override { return is_pins_set; }
101
102 protected:
103 arduino::HardwareGPIO_RPI* gpio = nullptr;
104 T lastValuePercent = 0.0f;
105 int pinStep = -1;
106 int pinDir = -1;
107 int pinEnable = -1;
108 bool is_pins_set = false;
109 float maxSpeedHz = 1000;
110 uint16_t accelerationMs = 0;
111 int stepsPerRevolution = 200;
112};
